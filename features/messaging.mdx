## Overview

CALMA's messaging system provides real-time communication between employers (Contratantes) and accepted candidates (Aspirantes) using WebSocket technology with the STOMP protocol.

## Chat Component

Location: `src/App.jsx`

The chat system is implemented in the main App component and provides:

- Real-time bidirectional messaging
- Message history persistence
- Online/offline status indicators
- Typing indicators
- Message notifications
- Read receipts

## WebSocket Connection

The chat system establishes a WebSocket connection using STOMP.js:

```javascript App.jsx lines
import { Client } from "@stomp/stompjs";

const cliente = new Client({
  brokerURL: "ws://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/ws",
  debug: str => console.log("📡", str),
  reconnectDelay: 5000,
  heartbeatIncoming: 4000,
  heartbeatOutgoing: 4000,
});

cliente.onConnect = () => {
  console.log("✅ WebSocket Conectado");
  setConectado(true);
  
  // Subscribe to conversation channel
  cliente.subscribe(canalConversacion, (msg) => {
    const data = JSON.parse(msg.body);
    handleIncomingMessage(data);
  });
  
  // Subscribe to personal notifications
  cliente.subscribe(`/tema/notificacion/${usuarioActual}`, (msg) => {
    const notificacion = JSON.parse(msg.body);
    handleNotification(notificacion);
  });
};

cliente.activate();
```

## Conversation Channel

Each conversation has a unique channel based on user IDs:

```javascript App.jsx:112-117
// Generate conversation channel
const canalConversacion = useMemo(() => {
  if (!usuarioActual || !destinatario) return null;
  const menor = Math.min(usuarioActual, destinatario);
  const mayor = Math.max(usuarioActual, destinatario);
  return `/tema/conversacion/${menor}-${mayor}`;
}, [usuarioActual, destinatario]);
```

This ensures:
- Unique channel per conversation pair
- Same channel regardless of who initiates
- Bidirectional message flow

## Sending Messages

Messages are sent through the STOMP client:

```javascript App.jsx:325-400
const enviarMensaje = useCallback(async () => {
  const contenido = mensaje.trim();
  if (!stompRef.current?.connected || !contenido) {
    return;
  }
  
  // Validate user IDs
  const validacion = validarYNormalizarIds(nombrePropio, destinatarioProp, 'ENVIO_MENSAJE');
  if (!validacion.valido) {
    alert(`Error de validación: ${validacion.error}`);
    return;
  }
  
  const { usuarioActualId, destinatarioId } = validacion;
  
  const mensajeData = {
    nombre: String(usuarioActualId),
    contenido,
    color: "#1976d2",
    aspiranteId: usuarioActualId,
    contratistaId: destinatarioId,
    remitenteId: usuarioActualId,
    fechaEnvio: new Date().toISOString(),
    tipo: "MENSAJE_CHAT",
    sessionId: sessionStorage.getItem('sessionId')
  };
  
  try {
    // Send via WebSocket
    stompRef.current.publish({
      destination: "/app/envio",
      body: JSON.stringify(mensajeData),
      headers: {
        'content-type': 'application/json',
        'priority': 'high'
      }
    });
    
    // Send notification
    await enviarNotificacionMensaje(
      Number(usuarioActualId),
      Number(destinatarioId),
      contenido
    );
    
    setMensaje("");
  } catch (error) {
    console.error("Error al enviar mensaje:", error);
    alert("Error al enviar el mensaje. Por favor intenta nuevamente.");
  }
}, [mensaje, nombrePropio, destinatarioProp]);
```

## Message History

Chat history is loaded when opening a conversation:

```javascript App.jsx:200-232
const cargarHistorial = async () => {
  try {
    const response = await fetch(
      `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/chat/historial?aspiranteId=${usuarioActual}&contratistaId=${destinatario}`
    );
    
    if (response.ok) {
      const data = await response.json();
      
      // Remove duplicates
      mensajesIds.current.clear();
      const mensajesUnicos = data.filter(m => {
        if (!m.id) return false;
        if (mensajesIds.current.has(m.id)) return false;
        mensajesIds.current.add(m.id);
        return true;
      });
      
      setMensajes(mensajesUnicos);
    }
  } catch (e) {
    console.error("Error cargando historial:", e);
    setMensajes([]);
  }
};
```

## Message Notifications

Separate API call for sending notifications:

```javascript App.jsx:128-191
const enviarNotificacionMensaje = useCallback(async (remitenteId, destinatarioId, contenido) => {
  try {
    // Verify IDs are numbers
    if (typeof remitenteId !== 'number' || typeof destinatarioId !== 'number') {
      throw new Error(`IDs must be numbers`);
    }
    
    const payload = {
      remitenteId: remitenteId,
      destinatarioId: destinatarioId,
      contenido: contenido,
      tipo: "MENSAJE_DIRECTO"
    };
    
    const response = await axios.post(
      'http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones/crear-notificacion-mensaje',
      payload,
      {
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        timeout: 10000
      }
    );
    
    if (response.status >= 200 && response.status < 300) {
      console.log('✅ Notificación enviada exitosamente');
      return true;
    }
    
    return false;
  } catch (error) {
    console.error('❌ Error enviando notificación:', error);
    return false;
  }
}, []);
```

## Message Validation

Strict ID validation prevents message routing errors:

```javascript App.jsx:31-84
const validarYNormalizarIds = useCallback((nombrePropio, destinatarioProp, contexto = '') => {
  const usuarioActualId = normalizarId(nombrePropio);
  const destinatarioId = normalizarId(destinatarioProp);
  
  // Critical validations
  if (usuarioActualId === null || destinatarioId === null) {
    return {
      valido: false,
      error: 'IDs cannot be null',
      usuarioActualId: null,
      destinatarioId: null
    };
  }
  
  if (isNaN(usuarioActualId) || isNaN(destinatarioId)) {
    return {
      valido: false,
      error: 'IDs must be valid numbers',
      usuarioActualId,
      destinatarioId
    };
  }
  
  if (usuarioActualId <= 0 || destinatarioId <= 0) {
    return {
      valido: false,
      error: 'IDs must be positive',
      usuarioActualId,
      destinatarioId
    };
  }
  
  if (usuarioActualId === destinatarioId) {
    return {
      valido: false,
      error: 'User cannot chat with themselves',
      usuarioActualId,
      destinatarioId
    };
  }
  
  return {
    valido: true,
    usuarioActualId: Number(usuarioActualId),
    destinatarioId: Number(destinatarioId),
    error: null
  };
}, [normalizarId]);
```
