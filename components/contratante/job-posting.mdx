### Validation

```javascript
const validarCampos = () => {
  const erroresTemp = {};
  const soloLetras = /^[A-Za-zÁÉÍÓÚáéíóúÑñ\s]+$/;
  const letrasNumeros = /^[A-Za-z0-9ÁÉÍÓÚáéíóúÑñ.,:;()\-_\s]+$/;
  const soloNumeros = /^\d+$/;

  if (!idPaciente) erroresTemp.idPaciente = 'Debe seleccionar un paciente.';
  
  if (!titulo.trim()) {
    erroresTemp.titulo = 'El título es obligatorio.';
  } else if (!soloLetras.test(titulo.trim())) {
    erroresTemp.titulo = 'El título solo debe contener letras.';
  }
  
  if (!fechaLimite) {
    erroresTemp.fechaLimite = 'Debe ingresar la fecha límite.';
  } else {
    const fechaSeleccionada = new Date(fechaLimite + 'T00:00:00');
    // Validate date range (today to +20 days)
  }
  
  if (!salarioEstimado) {
    erroresTemp.salarioEstimado = 'Campo obligatorio.';
  } else if (parseInt(salarioEstimado, 10) < 230) {
    erroresTemp.salarioEstimado = 'El salario no puede ser menor a $230.';
  }
  
  setErrores(erroresTemp);
  return Object.keys(erroresTemp).length === 0;
};
```

### Form Submission

```javascript
const handleSubmit = async (e) => {
  e.preventDefault();
  
  if (!validarCampos()) {
    toast.error('Por favor corrige los errores en el formulario.');
    return;
  }
  
  const fechaEnviar = fechaLimite ? new Date(fechaLimite + 'T00:00:00') : null;
  
  const data = {
    titulo,
    descripcion,
    fecha_limite: fechaEnviar,
    jornada,
    salario_estimado: salarioEstimado,
    requisitos,
    turno,
    estado,
    disponibilidad_inmediata: disponibilidadInmediata,
    actividades_realizar: actividadesRealizar,
    parroquia: { id_parroquia: idParroquia },
    id_paciente: Number(idPaciente),
  };
  
  try {
    if (publicacionEditar) {
      await axios.put(
        `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/publicacion_empleo/actualizar/${publicacionEditar.id_postulacion_empleo}`,
        data
      );
      toast.success('Publicación actualizada correctamente');
    } else {
      await axios.post(
        `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/publicacion_empleo/guardar?idParroquia=${idParroquia}&idContratante=${contratanteId}`,
        data
      );
      toast.success('Publicación creada correctamente');
      limpiarFormulario();
    }
    
    setTimeout(() => {
      navigate(`/moduloContratante/ListaPublicaciones?userId=${contratanteId}`);
    }, 1500);
  } catch (error) {
    toast.error('Error al guardar la publicación');
  }
};
```

### Filtering

```javascript
const publicacionesFiltradas = publicaciones.filter(pub => {
  const fecha = pub.publicacionempleo?.fecha_publicacion;
  const fechaNormalizada = fecha ? formatoLocal(fecha) : '';
  const coincideFecha = filtroFecha ? fechaNormalizada === filtroFecha : true;
  const coincideTitulo = pub.publicacionempleo?.titulo?.toLowerCase().includes(filtroTitulo.toLowerCase());
  return coincideFecha && coincideTitulo;
});
```

### Delete with Confirmation

```javascript
const mostrarConfirmacionEliminacion = (idPublicacion) => {
  const toastId = toast.info(
    <div>
      <p>¿Seguro que deseas eliminar esta publicación?</p>
      <button onClick={() => {
        toast.dismiss(toastId);
        axios.delete(`http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/publicacion_empleo/eliminar/${idPublicacion}`)
          .then(res => {
            toast.success(res.data);
            setPublicaciones(prev =>
              prev.filter(pub => pub.publicacionempleo.id_postulacion_empleo !== idPublicacion)
            );
          })
      }}>Eliminar</button>
      <button onClick={() => toast.dismiss(toastId)}>Cancelar</button>
    </div>,
    { autoClose: false, closeButton: false }
  );
};
```

## API Endpoints

| Endpoint | Method | Purpose |
|----------|--------|--------|
| `/api/publicacion_empleo/guardar` | POST | Create job posting |
| `/api/publicacion_empleo/actualizar/{id}` | PUT | Update job posting |
| `/api/publicacion_empleo/eliminar/{id}` | DELETE | Delete job posting |
| `/api/generar/contratante/{id}` | GET | List all postings |
| `/api/publicacion_empleo/pacientes/contratante/{id}` | GET | Get employer's patients |
