# Shared Form Components

Reusable form components used throughout the CALMA platform.

## CertificadosForm Component

### Location

`src/components/Certificados/CertificadosForm.jsx`

### Purpose

Manage certifications for elderly care professionals (aspirants).

### State Structure

```javascript
const [formulario, setFormulario] = useState({
  id_certificado: null,
  nombre_certificado: "",
  nombre_institucion: "",
  fecha: new Date().toISOString().slice(0, 10),
  archivo: null,
  archivoNombre: "",
  archivoExistente: false,
  nombreArchivoExistente: "",
  isEditing: false
});

const [certificados, setCertificados] = useState([]);
const [isLoading, setIsLoading] = useState(false);
const [isSubmitting, setIsSubmitting] = useState(false);
```

### Form Fields

1. **nombre_certificado** (text, required)
   - Label: "Nombre de la Certificación"
   - Placeholder: "Ej: Cuidado de Adultos Mayores con Demencia"
   - Icon: `FaUserMd`

2. **nombre_institucion** (text, required)
   - Label: "Institución Certificadora"
   - Placeholder: "Ej: Cruz Roja, Instituto Geriátrico Nacional"
   - Icon: `FaUniversity`

3. **fecha** (date)
   - Label: "Fecha de Obtención"
   - Icon: `FaCalendarAlt`

4. **archivo** (file, required for new, optional for edit)
   - Label: "Certificado Digital"
   - Accept: `.pdf,.jpg,.png,.doc,.docx`
   - Icon: `FaPaperclip`

### Delete with Confirmation

```javascript
const manejarEliminar = async (id) => {
  const certificado = certificados.find(cert => cert.id_certificado === id);
  
  showConfirmModal(
    "Eliminar Certificación",
    `¿Estás seguro de que deseas eliminar "${certificado?.nombre_certificado}"?`,
    async () => {
      try {
        await deleteCertificado(id);
        setCertificados(certificados.filter(cert => cert.id_certificado !== id));
        toast.success("Certificación eliminada correctamente");
      } catch (error) {
        toast.error("Error al eliminar la certificación");
      }
      closeConfirmModal();
    }
  );
};
```

```javascript
const manejarEnvio = async (e) => {
  e.preventDefault();
  
  if (!formulario.descripcion.trim()) {
    toast.warning("Por favor describe una habilidad específica");
    return;
  }
  
  const datosHabilidad = {
    id_habilidad: formulario.id_habilidad,
    descripcion: formulario.descripcion,
    nivel: formulario.nivel,
    cv: { id_cv: Number(idCV) }
  };
  
  if (formulario.isEditing) {
    const habilidadActualizada = await updateHabilidad(formulario.id_habilidad, datosHabilidad);
    setHabilidades(habilidades.map(hab => 
      hab.id_habilidad === formulario.id_habilidad ? habilidadActualizada : hab
    ));
    toast.success("Habilidad actualizada");
  } else {
    const nuevaHabilidad = await createHabilidad(datosHabilidad);
    setHabilidades([...habilidades, nuevaHabilidad]);
    toast.success("Habilidad guardada");
  }
  
  reiniciarFormulario();
};
```

## RequireCV Component

### Purpose

Route protection component that ensures aspirants have completed their CV.

### Location

`src/components/ModuloAspirante/CV/RequireCV.jsx`

### Usage

```jsx
import RequireCV from './components/ModuloAspirante/CV/RequireCV';

<Route
  path="/moduloAspirante/*"
  element={
    <RequireCV>
      <ModuloAspirante />
    </RequireCV>
  }
/>
```

### Implementation

```javascript
const RequireCV = ({ children }) => {
  const userData = JSON.parse(localStorage.getItem('userData'));
  const aspiranteId = userData?.aspiranteId;
  const [hasCV, setHasCV] = useState(null);
  const [loading, setLoading] = useState(true);
  
  useEffect(() => {
    const checkCV = async () => {
      try {
        const tieneCV = await checkIfAspiranteHasCV(aspiranteId);
        setHasCV(tieneCV);
      } catch (error) {
        setHasCV(false);
      } finally {
        setLoading(false);
      }
    };
    
    if (aspiranteId) {
      checkCV();
    } else {
      setLoading(false);
    }
  }, [aspiranteId]);
  
  if (loading) return <div>Verificando perfil...</div>;
  
  if (!hasCV) {
    return <Navigate to="/moduloAspirante/cv" replace />;
  }
  
  return children;
};
```

## Form Persistence Hook

### useFormPersistence

```javascript
import { useEffect } from 'react';

export const useFormPersistence = (key, formData, setFormData, formName) => {
  const storageKey = `form_${formName}_${key}`;
  
  // Save to localStorage on change
  useEffect(() => {
    if (formData && !formData.isEditing) {
      localStorage.setItem(storageKey, JSON.stringify(formData));
    }
  }, [formData, storageKey]);
  
  // Load from localStorage on mount
  useEffect(() => {
    const savedData = localStorage.getItem(storageKey);
    if (savedData) {
      try {
        const parsed = JSON.parse(savedData);
        setFormData(parsed);
      } catch (e) {
        console.error('Error parsing saved form data:', e);
      }
    }
  }, [storageKey, setFormData]);
  
  // Clear on successful submission
  const clearPersistedData = () => {
    localStorage.removeItem(storageKey);
  };
  
  return { clearPersistedData };
};
```

### Usage

```javascript
import { useFormPersistence } from '../../hooks/useFormPersistence';

const CertificadosForm = () => {
  const { idCV } = useParams();
  const [formulario, setFormulario] = useState({ /* ... */ });
  
  useFormPersistence(idCV, formulario, setFormulario, 'certificados');
  
  // Form logic...
};
```
