## Overview

The CV API manages caregiver resumes (CVs), including personal information, work experience, certifications, skills, and availability.

## Base URL

```
http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/cvs
```

## Get All CVs

### GET /api/cvs

Retrieve all CVs in the system.

```javascript
import { getCVs } from './servicios/cvService';

try {
  const cvs = await getCVs();
  console.log('All CVs:', cvs);
} catch (error) {
  console.error('Error:', error.message);
}
```

### Service Implementation

```javascript
export const getCVs = async () => {
  try {
    const response = await axios.get(
      'http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/cvs'
    );
    return response.data;
  } catch (error) {
    console.error('Error al obtener todos los CVs:', error);
    throw new Error('No se pudieron cargar los CVs');
  }
};
```

export const getCVById = async (id) => {
  try {
    console.log(`Obteniendo CV con ID: ${id}`);

      `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/cvs/${id}`

    return response.data;

      if (error.response.status === 404) {
        errorMessage = 'El CV solicitado no existe';

        errorMessage = 'Error en el servidor al procesar tu solicitud';
      }
      console.error('Detalles del error:', error.response.data);
    }
    
    throw new Error(errorMessage);
  }
};

## Get CV by Aspirante ID

### GET /api/cvs/por-aspirante/{aspiranteId}

Retrieve a CV by the aspirante (caregiver) ID.

**Location:** `source/src/servicios/cvService.js:87`

```javascript
import { getCVByAspiranteId } from './servicios/cvService';

try {
  const cv = await getCVByAspiranteId(15);
  if (cv) {
    console.log('CV found:', cv);
  } else {
    console.log('No CV found for this aspirante');
  }
} catch (error) {
  console.error('Error:', error.message);
}
```

### Service Implementation

```javascript
export const getCVByAspiranteId = async (aspiranteId) => {
  try {
    const response = await axios.get(
      `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/cvs/por-aspirante/${aspiranteId}`
    );
    
    // Validate response data
    if (!response.data || !response.data.id_cv) {
      console.log(`No se encontró CV para el aspirante ${aspiranteId}`);
      return null;
    }
    
    // Validate CV belongs to the requested aspirante
    if (response.data.aspirante?.idAspirante?.toString() !== aspiranteId?.toString()) {
      console.error('El CV no pertenece al aspirante solicitado');
      return null;
    }
    
    return response.data;
  } catch (error) {
    console.error(`Error al obtener CV para aspirante ${aspiranteId}:`, error);
    throw new Error('Error al verificar el CV');
  }
};
```

## Error Handling

The service includes comprehensive error handling:

```javascript
try {
  const cv = await getCVById(id);
  return cv;
} catch (error) {
  if (error.response) {
    // Server responded with error
    if (error.response.status === 404) {
      throw new Error('El CV solicitado no existe');
    } else if (error.response.status === 500) {
      throw new Error('Error en el servidor al procesar tu solicitud');
    }
  } else if (error.request) {
    // Request made but no response
    throw new Error('No se pudo conectar con el servidor');
  } else {
    // Other errors
    throw new Error('Error inesperado');
  }
}
```

## Usage in Components

### Check CV Exists Before Creating

```javascript
import { checkIfAspiranteHasCV, getCVByAspiranteId, createCV } from './servicios/cvService';

const handleCreateCV = async (aspiranteId, cvData) => {
  // Check if CV already exists
  const hasCV = await checkIfAspiranteHasCV(aspiranteId);
  
  if (hasCV) {
    // Get existing CV
    const existingCV = await getCVByAspiranteId(aspiranteId);
    console.log('Aspirante already has CV:', existingCV.id_cv);
    
    // Redirect to edit instead
    navigate(`/cv/editar/${existingCV.id_cv}`);
  } else {
    // Create new CV
    const newCV = await createCV({
      ...cvData,
      aspirante: { idAspirante: aspiranteId }
    });
    
    console.log('CV created successfully');
    navigate(`/cv/${newCV.id_cv}`);
  }
};
```

### Update CV with Validation

```javascript
const handleUpdateCV = async (cvId, aspiranteId, updatedData) => {
  try {
    // Validate ownership
    const existingCV = await getCVById(cvId);
    
    if (existingCV.aspirante.idAspirante !== aspiranteId) {
      throw new Error('No tienes permiso para editar este CV');
    }
    
    // Update CV
    const updated = await updateCV(cvId, updatedData);
    toast.success('CV actualizado exitosamente');
    return updated;
  } catch (error) {
    toast.error(error.message);
  }
};
```
