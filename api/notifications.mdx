## Overview

The Notifications API manages system notifications for both caregivers (aspirantes) and contractors (contratantes). Notifications inform users about application statuses, messages, and system updates.

## Base URL

```
http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones
```

## Get Unread Notifications (Aspirante)

### GET /api/notificaciones/aspirante/noleidas/{idAspirante}

Retrieve count of unread notifications for a caregiver.

**Location:** Used in **source/src/components/ModuloAspirante/HeaderAspirante/headerAspirante.jsx**

```javascript
const response = await axios.get(
  `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones/aspirante/noleidas/${idAspirante}`
);

const unreadCount = response.data; // Returns number
```

<ResponseField name="response.data" type="number">
  Count of unread notifications
</ResponseField>

## Implementation in Header

### Aspirante Header

**Location:** `source/src/components/ModuloAspirante/HeaderAspirante/headerAspirante.jsx`

```javascript
import { useState, useEffect } from 'react';
import axios from 'axios';

const HeaderAspirante = ({ aspiranteId }) => {
  const [notificaciones, setNotificaciones] = useState([]);
  const [notificacionesNoLeidas, setNotificacionesNoLeidas] = useState(0);

  // Load unread count
  useEffect(() => {
    const cargarNotificacionesNoLeidas = async () => {
      try {
        const response = await axios.get(
          `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones/aspirante/noleidas/${aspiranteId}`
        );
        setNotificacionesNoLeidas(response.data);
      } catch (error) {
        console.error('Error cargando notificaciones:', error);
      }
    };

    if (aspiranteId) {
      cargarNotificacionesNoLeidas();
      // Poll every 30 seconds
      const interval = setInterval(cargarNotificacionesNoLeidas, 30000);
      return () => clearInterval(interval);
    }
  }, [aspiranteId]);

  // Load all notifications
  const cargarNotificaciones = async () => {
    try {
      const response = await axios.get(
        `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones/aspirante/${aspiranteId}`
      );
      setNotificaciones(response.data);
    } catch (error) {
      console.error('Error:', error);
    }
  };

  // Mark all as read
  const marcarComoLeidas = async () => {
    try {
      await axios.put(
        `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/notificaciones/aspirante/marcar-leidas/${aspiranteId}`
      );
      setNotificacionesNoLeidas(0);
      // Reload notifications
      await cargarNotificaciones();
    } catch (error) {
      console.error('Error marcando como leídas:', error);
    }
  };

  return (
    <header>
      <button onClick={cargarNotificaciones}>
        <i className="bell-icon"></i>
        {notificacionesNoLeidas > 0 && (
          <span className="badge">{notificacionesNoLeidas}</span>
        )}
      </button>
    </header>
  );
};
```

## Date Formatting

```javascript
const formatNotificationDate = (fecha) => {
  const now = new Date();
  const notifDate = new Date(fecha);
  const diff = now - notifDate;
  const minutes = Math.floor(diff / 60000);
  const hours = Math.floor(diff / 3600000);
  const days = Math.floor(diff / 86400000);

  if (minutes < 1) return 'Ahora';
  if (minutes < 60) return `Hace ${minutes} min`;
  if (hours < 24) return `Hace ${hours} h`;
  if (days < 7) return `Hace ${days} d`;
  
  return notifDate.toLocaleDateString('es-ES', {
    month: 'short',
    day: 'numeric'
  });
};
```

## Polling Strategy

Notifications are typically polled every 30 seconds:

```javascript
useEffect(() => {
  const loadNotifications = async () => {
    // Load logic
  };

  loadNotifications(); // Initial load
  
  const interval = setInterval(loadNotifications, 30000); // Poll every 30s
  
  return () => clearInterval(interval); // Cleanup
}, [userId]);
```

<Tip>
  For real-time notifications, consider implementing WebSocket connections instead of polling.
</Tip>

## Usage Examples

### Display Notification List

```javascript
const NotificationList = ({ notificaciones, onMarkAsRead }) => {
  return (
    <div className="notification-list">
      {notificaciones.map(notif => (
        <div 
          key={notif.id_notificacion}
          className={`notification-item ${!notif.leida ? 'unread' : ''}`}
        >
          <div className="notification-icon">
            {getIconForType(notif.tipo)}
          </div>
          <div className="notification-content">
            <p>{notif.mensaje}</p>
            <span className="notification-date">
              {formatNotificationDate(notif.fecha)}
            </span>
          </div>
          {!notif.leida && (
            <span className="unread-dot"></span>
          )}
        </div>
      ))}
      
      {notificaciones.some(n => !n.leida) && (
        <button onClick={onMarkAsRead}>
          Marcar todas como leídas
        </button>
      )}
    </div>
  );
};

const getIconForType = (tipo) => {
  switch (tipo) {
    case 'postulacion': return '💼';
    case 'trabajo': return '👥';
    case 'mensaje': return '💬';
    case 'sistema': return '⚙️';
    default: return '🔔';
  }
};
```
