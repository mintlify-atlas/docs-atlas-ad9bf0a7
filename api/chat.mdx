### Sending Questions

```javascript
const enviarPregunta = async () => {
  if (!pregunta.trim()) return;

  // Add user message to conversation
  setConversacion((prev) => [...prev, { from: 'user', text: pregunta }]);
  setPregunta('');
  setCargando(true);
  setError(null);

  try {
    // Check for special keywords
    if (
      pregunta.toLowerCase().includes('postular') ||
      pregunta.toLowerCase().includes('registrar')
    ) {
      // Hardcoded response for registration questions
      setConversacion((prev) => [
        ...prev,
        {
          from: 'bot',
          text: 'Para postularte a Calma, primero debes registrarte en nuestra plataforma. Puedes hacerlo desde aquí 👉 <a href="/registro">Ir al registro</a>',
        },
      ]);
    } else {
      // Call chatbot API
      const res = await axios.post(
        'http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/chatbot/preguntar',
        { pregunta }
      );
      
      if (res.data && res.data.respuesta) {
        setConversacion((prev) => [
          ...prev,
          { from: 'bot', text: res.data.respuesta }
        ]);
      } else {
        setConversacion((prev) => [
          ...prev,
          { from: 'bot', text: '⚠️ No se pudo obtener una respuesta válida del chatbot.' }
        ]);
      }
    }
  } catch (e) {
    console.error(e);
    setError('Error al comunicarse con el chatbot.');
    setConversacion((prev) => [
      ...prev,
      { from: 'bot', text: '⚠️ Hubo un error al obtener respuesta.' }
    ]);
  } finally {
    setCargando(false);
  }
};
```

## Special Response Formatting

The chatbot can return formatted job listings:

```javascript
const renderBotMessage = (text) => {
  // Check if response contains job offer separators
  if (text.includes('━━━━━━━━━━━━━━━━━━━━━━')) {
    const ofertas = text.split('━━━━━━━━━━━━━━━━━━━━━━')
      .filter((o) => o.trim() !== '');
    
    return (
      <>
        <div className="bot-intro-bubble">
          🤖 Estas son algunas oportunidades activas hoy:
        </div>
        {ofertas.map((oferta, idx) => (
          <div
            key={idx}
            className="chat-message bot oferta-bubble"
            dangerouslySetInnerHTML={{ 
              __html: oferta.trim().replace(/\n/g, '<br />') 
            }}
          />
        ))}
      </>
    );
  }
  
  // Render with markdown-style links
  return (
    <span
      dangerouslySetInnerHTML={{
        __html: text.replace(
          /\[(.*?)\]\((.*?)\)/g,
          '<a href="$2" target="_blank" rel="noopener noreferrer">$1</a>'
        ),
      }}
    />
  );
};
```

## Usage Example: Intelligent Chatbot

**Location:** `source/src/components/ModuloAspirante/PostulacionesAspirante/ChatbotInteligente.jsx`

```javascript
import { useState } from 'react';
import axios from 'axios';

const ChatbotInteligente = ({ aspiranteId }) => {
  const [abierto, setAbierto] = useState(false);
  const [pregunta, setPregunta] = useState('');
  const [conversacion, setConversacion] = useState([]);

  const enviarPregunta = async () => {
    if (!pregunta.trim()) return;

    // Add user message
    setConversacion(prev => [
      ...prev,
      { from: 'user', text: pregunta }
    ]);

    try {
      // Get aspirante's applications for context
      const postulaciones = await axios.get(
        `http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/realizar/aspirante/${aspiranteId}`
      );

      // Send to chatbot with context
      const response = await axios.post(
        'http://backend-alb-283290471.us-east-2.elb.amazonaws.com:8090/api/chatbot/preguntar',
        {
          pregunta,
          contexto: {
            postulaciones: postulaciones.data.length,
            aspiranteId
          }
        }
      );

      // Add bot response
      setConversacion(prev => [
        ...prev,
        { from: 'bot', text: response.data.respuesta }
      ]);
    } catch (error) {
      console.error('Error:', error);
    }

    setPregunta('');
  };

  return (
    <div className="chatbot-container">
      {/* Chatbot UI */}
    </div>
  );
};
```
